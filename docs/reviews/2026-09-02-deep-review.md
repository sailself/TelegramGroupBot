```rust
) -> Result<Value, ProviderError>;   // logs started, retries, records success *and* failure

pub struct ToolCall { id: Option<String>, name: String, args: Value }
pub struct ParsedTurn<T> { text: String, tool_calls: Vec<ToolCall>, assistant_turn: T }

pub trait ChatProvider {
    type Turn: Clone;   // Gemini `contents` item / chat message / responses input item
    fn initial_turns(&self, system: &str, user: &str, media: &[MediaFile]) -> Result<Vec<Self::Turn>>;
    fn tool_declarations(&self, rt: &ToolRuntime) -> Vec<Value>;
    fn build_request(&self, turns: &[Self::Turn], tools: Option<&[Value]>, final_schema: Option<&Value>) -> HttpRequestSpec;
    fn parse_turn(&self, response: &Value) -> Result<ParsedTurn<Self::Turn>>;
    fn tool_result_turns(&self, calls: &[(ToolCall, String)]) -> Vec<Self::Turn>;
    fn tool_limit_turn(&self) -> Option<Self::Turn>;   // None for Gemini
    fn usage(&self, response: &Value) -> LlmUsageRecord;
    fn retry_policy(&self) -> RetryPolicy;
    fn envelope(&self, operation: &str) -> CallEnvelope<'_>;
}

pub async fn run_tool_loop<P: ChatProvider>(p: &P, rt: &mut ToolRuntime, system: &str, user: &str,
    media: &[MediaFile], final_schema: Option<&Value>, audit: Option<&LlmAuditContext>) -> Result<String>;
```
